            <h3 class="title"> VNM (Virtual Net Metering) </h3>
            <div class="subtxt">Virtual Net Metering System</div>
            <ul class="list">
                <li class="item">
                    <div class="num">01</div>
                    <div class="tit">Remote Solar Power Generation</div>
                    <div class="detail"> Electricity generated at a solar power plant <br> located in another area
                    </div>
                </li>
                <p class="next"><i class="xi-arrow-right"></i></p>
                <li class="item">
                    <div class="num">02</div>
                    <div class="tit">Grid Connection</div>
                    <div class="detail"> Generated power transmitted <br> to the KEPCO grid </div>
                </li>
                <p class="next"><i class="xi-arrow-right"></i></p>
                <li class="item">
                    <div class="num">03</div>
                    <div class="tit">Power Metering (kWh)</div>
                    <div class="detail"> Measurement of generated electricity </div>
                </li>
                <p class="next"><i class="xi-arrow-right"></i></p>
                <li class="item">
                    <div class="num">04</div>
                    <div class="tit">Conversion to Credit</div>
                    <div class="detail"> kWh converted into billing credits </div>
                </li>
                <p class="next"><i class="xi-arrow-right"></i></p>
                <li class="item">
                    <div class="num">05</div>
                    <div class="tit">Credit Accumulation</div>
                    <div class="detail"> Credits stored in the subscriber’s account </div>
                </li>
                <p class="next"><i class="xi-arrow-right"></i></p>
                <li class="item">
                    <div class="num">06</div>
                    <div class="tit">Electricity Bill Deduction</div>
                    <div class="detail"> Credits applied to reduce bills <br> of another building </div>
                </li>
            </ul>
